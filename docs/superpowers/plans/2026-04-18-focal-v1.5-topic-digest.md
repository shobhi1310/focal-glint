# Focal v1.5 — Topic-Based Digest Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Replace the 3-class system with 4-class (Urgent/Actionable/Digest/Noise), add topic clustering with cross-app semantic deduplication, and redesign the Digest as a topic-based "news shorts" feed with drill-down detail views.

**Architecture:** Topic clustering runs as a new step in ClassificationWorker after classification + summarization. TopicEngine groups notifications by app, uses LLM to merge semantically similar groups across apps, then generates detail views (template-based for billing/delivery/calendar, LLM-generated for everything else). Topics are stored in a new Room table and served to a rewritten Digest UI via TopicRepository.

**Tech Stack:** Room (migration v1→v2), Kotlin Coroutines, Jetpack Compose, Hilt, LiteRT-LM, WorkManager

            classifiedBy = "llm",
            confidence = confidence,

        )

abstract class FocalDatabase : RoomDatabase() {

    abstract fun correctionDao(): CorrectionDao

                // Create topics table

                    CREATE TABLE IF NOT EXISTS topics (
                        id TEXT PRIMARY KEY NOT NULL,
                        headline TEXT NOT NULL,
                        summary TEXT NOT NULL,
                        category TEXT NOT NULL,
                        notification_ids TEXT NOT NULL,
                        source_apps TEXT NOT NULL,

                db.execSQL("CREATE INDEX IF NOT EXISTS index_topics_updated_at ON topics(updated_at)")

                // Add topic_id to notifications
                db.execSQL("ALTER TABLE notifications ADD COLUMN topic_id TEXT")

                // Migrate informational → digest
                db.execSQL("UPDATE notifications SET category = 'digest' WHERE category = 'informational'")

                // Migrate rules
                db.execSQL("""
                    UPDATE rules SET category = 'digest' WHERE category = 'informational' AND app IN (

                db.execSQL("""
                    UPDATE rules SET category = 'actionable' WHERE category = 'informational' AND app IN (
                        'in.swiggy.android', 'com.application.zomato'

// app/src/main/java/com/focal/intelligence/DetailTemplates.kt
package com.focal.intelligence

import com.focal.data.db.entity.NotificationEntity
import org.json.JSONObject

object DetailTemplates {

    fun tryExtractDetail(notifications: List<NotificationEntity>, appType: String): Pair<String?, String?> {
        return when (appType) {
            "billing" -> extractBilling(notifications)
            "transactional" -> extractDelivery(notifications)
            "calendar" -> extractCalendar(notifications)
            else -> null to null
        }
    }

    fun detectAppType(appName: String, packageName: String): String {
        val name = appName.lowercase()
        val pkg = packageName.lowercase()
        return when {
            pkg.contains("cred") || name.contains("cred") -> "billing"
            pkg.contains("swiggy") || pkg.contains("zomato") || pkg.contains("amazon") -> "transactional"
            pkg.contains("calendar") || name.contains("calendar") -> "calendar"
            pkg.contains("whatsapp") || pkg.contains("telegram") || pkg.contains("slack") || pkg.contains("discord") -> "messaging"
            else -> "general"
        }
    }

    private fun extractBilling(notifications: List<NotificationEntity>): Pair<String?, String?> {
        val allText = notifications.joinToString(" ") { "${it.title} ${it.content} ${it.bigText ?: ""}" }

        val amountRegex = Regex("[₹Rs\\.]+\\s*([\\d,]+\\.?\\d*)")
        val amount = amountRegex.find(allText)?.groupValues?.get(1)?.replace(",", "")

        val dueDateRegex = Regex("(?:due|by|before)\\s+(?:on\\s+)?([A-Za-z]+\\s+\\d{1,2}(?:,?\\s*\\d{4})?)", RegexOption.IGNORE_CASE)
        val dueDate = dueDateRegex.find(allText)?.groupValues?.get(1)

        val cardRegex = Regex("(?:card|ending)\\s*(?:in\\s+|[-x]*)?\\s*(\\d{4})", RegexOption.IGNORE_CASE)
        val cardLast4 = cardRegex.find(allText)?.groupValues?.get(1)

        if (amount == null && dueDate == null) return null to null

        val json = JSONObject().apply {
            put("type", "billing")
            amount?.let { put("amount", it) }
            put("currency", "INR")
            dueDate?.let { put("due_date", it) }
            cardLast4?.let { put("card_last4", it) }
            put("status", "unpaid")
        }

        val label = if (cardLast4 != null) "Open CRED" else null

        return json.toString() to label
    }

    private fun extractDelivery(notifications: List<NotificationEntity>): Pair<String?, String?> {
        val latest = notifications.maxByOrNull { it.postedAt } ?: return null to null
        val text = "${latest.title} ${latest.content} ${latest.bigText ?: ""}".lowercase()

        val status = when {
            text.contains("delivered") -> "delivered"
            text.contains("out for delivery") -> "out_for_delivery"
            text.contains("preparing") || text.contains("being prepared") -> "preparing"
            text.contains("confirmed") || text.contains("placed") -> "ordered"
            else -> "unknown"
        }

        val json = JSONObject().apply {
            put("type", "delivery")
            put("status", status)
            put("latest_message", latest.content.take(100))
        }

        return json.toString() to null
    }

    private fun extractCalendar(notifications: List<NotificationEntity>): Pair<String?, String?> {
        val latest = notifications.maxByOrNull { it.postedAt } ?: return null to null
        val text = "${latest.title} ${latest.content} ${latest.bigText ?: ""}"

        val eventRegex = Regex("(?:until this event:|Reminder:|Event:)\\s*(.+)", RegexOption.IGNORE_CASE)
        val event = eventRegex.find(text)?.groupValues?.get(1)?.trim() ?: latest.content

        val json = JSONObject().apply {
            put("type", "calendar")
            put("event", event.take(100))
        }

        return json.toString() to null
    }
}

// app/src/main/java/com/focal/intelligence/TopicEngine.kt
package com.focal.intelligence

import android.util.Log
import com.focal.data.db.entity.NotificationEntity
import com.focal.data.db.entity.TopicEntity
import com.focal.data.repository.NotificationRepository
import com.focal.data.repository.TopicRepository
import org.json.JSONArray

class TopicEngine(
    private val inferenceProvider: InferenceProvider,
    private val notificationRepository: NotificationRepository,
    private val topicRepository: TopicRepository,
    private val summarizer: Summarizer
) {
    suspend fun generateTopics() {
        val since = System.currentTimeMillis() - 24 * 60 * 60 * 1000L
        val allNotifications = notificationRepository.getRecentNotificationsSnapshot()
            .filter { !it.isSummary && it.category != ClassificationResult.UNCATEGORIZED }

        if (allNotifications.isEmpty()) return

        // Step 1: Group by app
        val appGroups = allNotifications.groupBy { it.packageName }

        // Step 2: Create initial topic per app group
        val initialTopics = appGroups.map { (packageName, notifications) ->
            val appName = notifications.first().appName
            val appType = DetailTemplates.detectAppType(appName, packageName)
            val category = notifications.maxByOrNull { it.postedAt }?.category ?: ClassificationResult.DIGEST

            val summary = summarizer.summarizeForApp(packageName)

            AppTopic(
                packageName = packageName,
                appName = appName,
                appType = appType,
                notifications = notifications,
                headline = appName,
                summary = summary?.summaryText ?: "${notifications.size} notifications from $appName",
                category = category
            )
        }

        // Step 3: LLM merge pass (semantic dedup)
        val mergedTopics = if (inferenceProvider.isReady() && initialTopics.size > 1) {
            mergeTopics(initialTopics)
        } else {
            initialTopics.map { listOf(it) }
        }

        // Step 4: Generate detail views and build TopicEntities
        val topicEntities = mergedTopics.mapNotNull { group ->
            buildTopicEntity(group)
        }

        // Step 5: Save to DB
        topicRepository.clearAndSaveTopics(topicEntities)
        Log.d("TopicEngine", "Generated ${topicEntities.size} topics from ${allNotifications.size} notifications")
    }

    private suspend fun mergeTopics(topics: List<AppTopic>): List<List<AppTopic>> {
        val prompt = buildMergePrompt(topics)
        return try {
            val response = inferenceProvider.generate(prompt, maxTokens = 256)
            parseMergeResponse(response, topics)
        } catch (e: Exception) {
            Log.w("TopicEngine", "LLM merge failed, falling back to no-merge", e)
            topics.map { listOf(it) }
        }
    }

    private fun buildMergePrompt(topics: List<AppTopic>): String {
        val sb = StringBuilder()
        sb.appendLine("Here are notification groups from the last 24 hours:")
        topics.forEachIndexed { index, topic ->
            sb.appendLine("${index + 1}. ${topic.appName}: ${topic.summary.take(80)}")
        }
        sb.appendLine()
        sb.appendLine("Which groups are about the same topic? Respond as JSON array of arrays:")
        sb.appendLine("Example: [[1,2],[3],[4,5]] means groups 1&2 are the same topic, 3 is standalone, 4&5 are the same.")
        return sb.toString()
    }

    private fun parseMergeResponse(response: String, topics: List<AppTopic>): List<List<AppTopic>> {
        return try {
            val jsonStr = response.trim().let { raw ->
                val start = raw.indexOf('[')
                val end = raw.lastIndexOf(']')
                if (start >= 0 && end > start) raw.substring(start, end + 1) else null
            } ?: return topics.map { listOf(it) }

            val outerArray = JSONArray(jsonStr)
            val result = mutableListOf<List<AppTopic>>()
            val used = mutableSetOf<Int>()

            for (i in 0 until outerArray.length()) {
                val innerArray = outerArray.getJSONArray(i)
                val group = mutableListOf<AppTopic>()
                for (j in 0 until innerArray.length()) {
                    val idx = innerArray.getInt(j) - 1  // 1-indexed to 0-indexed
                    if (idx in topics.indices && idx !in used) {
                        group.add(topics[idx])
                        used.add(idx)
                    }
                }
                if (group.isNotEmpty()) result.add(group)
            }

            // Add any ungrouped topics
            topics.forEachIndexed { index, topic ->
                if (index !in used) result.add(listOf(topic))
            }

            result
        } catch (e: Exception) {
            Log.w("TopicEngine", "Failed to parse merge response", e)
            topics.map { listOf(it) }
        }
    }

    private suspend fun buildTopicEntity(group: List<AppTopic>): TopicEntity? {
        if (group.isEmpty()) return null

        val allNotifications = group.flatMap { it.notifications }
        val allNotificationIds = allNotifications.map { it.id }
        val sourceApps = group.map { it.appName }.distinct()
        val primaryTopic = group.first()
        val category = determineCategoryForGroup(group)

        // Generate headline
        val headline = if (group.size == 1) {
            primaryTopic.headline
        } else {
            generateMergedHeadline(group)
        }

        // Generate summary
        val summary = if (group.size == 1) {
            primaryTopic.summary
        } else {
            group.joinToString(". ") { it.summary.take(60) }
        }

        // Generate detail view
        val (detailJson, actionLabel) = DetailTemplates.tryExtractDetail(
            allNotifications,
            primaryTopic.appType
        )

        val detailSummary = if (detailJson == null && inferenceProvider.isReady()) {
            try {
                val prompt = PromptBuilder.buildSummarizationPrompt(
                    appName = sourceApps.joinToString(" + "),
                    conversationName = null,
                    notifications = allNotifications.take(20)
                )
                val response = inferenceProvider.generate(prompt, maxTokens = 150)
                LlmResponseParser.parseSummary(response)
            } catch (e: Exception) {
                null
            }
        } else null

        return TopicEntity(
            headline = headline,
            summary = summary,
            category = category,
            notificationIds = JSONArray(allNotificationIds).toString(),
            sourceApps = JSONArray(sourceApps).toString(),
            channelCount = sourceApps.size,
            detailJson = detailJson,
            detailSummary = detailSummary,
            actionLabel = actionLabel,
            actionPackage = primaryTopic.packageName
        )
    }

    private fun determineCategoryForGroup(group: List<AppTopic>): String {
        val categories = group.map { it.category }
        return when {
            ClassificationResult.URGENT in categories -> ClassificationResult.URGENT
            ClassificationResult.ACTIONABLE in categories -> ClassificationResult.ACTIONABLE
            ClassificationResult.DIGEST in categories -> ClassificationResult.DIGEST
            else -> ClassificationResult.NOISE
        }
    }

    private fun generateMergedHeadline(group: List<AppTopic>): String {
        val summaries = group.map { it.summary }
        val combined = summaries.joinToString(" ").take(60)
        return combined.substringBefore(".").trim()
    }

    private data class AppTopic(
        val packageName: String,
        val appName: String,
        val appType: String,
        val notifications: List<NotificationEntity>,
        val headline: String,
        val summary: String,
        val category: String
    )
}

package com.focal.worker

import android.content.Context
import android.util.Log
import androidx.hilt.work.HiltWorker
import androidx.work.CoroutineWorker
import androidx.work.WorkerParameters
import com.focal.data.repository.NotificationRepository
import com.focal.intelligence.Classifier
import com.focal.intelligence.Summarizer
import com.focal.intelligence.TopicEngine
import dagger.assisted.Assisted
import dagger.assisted.AssistedInject

@HiltWorker
class ClassificationWorker @AssistedInject constructor(
    @Assisted appContext: Context,
    @Assisted workerParams: WorkerParameters,
    private val notificationRepository: NotificationRepository,
    private val classifier: Classifier,
    private val summarizer: Summarizer,
    private val topicEngine: TopicEngine
) : CoroutineWorker(appContext, workerParams) {

    override suspend fun doWork(): Result {
        Log.d("ClassificationWorker", "Starting batch processing")

        // Step 1: Classify pending notifications
        val pending = notificationRepository.getPendingForClassification()
        if (pending.isNotEmpty()) {
            Log.d("ClassificationWorker", "Classifying ${pending.size} pending notifications")
            var classified = 0
            for (notification in pending) {
                try {
                    val result = classifier.classify(notification)
                    notificationRepository.markClassified(
                        notification = notification,
                        category = result.category,
                        classifiedBy = result.classifiedBy
                    )
                    classified++
                } catch (e: Exception) {
                    Log.e("ClassificationWorker", "Failed to classify ${notification.id}", e)
                }
            }
            Log.d("ClassificationWorker", "Classified $classified/${pending.size}")
        }

        // Step 2: Generate topics (clustering + dedup + detail views)
        try {
            topicEngine.generateTopics()
            Log.d("ClassificationWorker", "Topic generation complete")
        } catch (e: Exception) {
            Log.e("ClassificationWorker", "Topic generation failed", e)
        }

        return Result.success()
    }

    companion object {
        const val WORK_NAME = "focal_classification"
    }
}

            ClassificationResult.URGENT -> UrgentRed
            ClassificationResult.ACTIONABLE -> ActionableAmber
            ClassificationResult.DIGEST -> DigestBlue
            else -> MaterialTheme.colorScheme.onSurface.copy(alpha = 0.5f)
        }
        val label = when (topic.category) {
            ClassificationResult.URGENT -> "🔴 URGENT"
            ClassificationResult.ACTIONABLE -> "🟡 ACTION NEEDED"
            ClassificationResult.DIGEST -> "🔵 DIGEST"
            else -> "⚫ NOISE"
        }

        Text(text = label, fontSize = 10.sp, fontWeight = FontWeight.Bold, color = labelColor)

        Text(
            text = topic.headline,
            style = MaterialTheme.typography.headlineLarge,
            color = MaterialTheme.colorScheme.onBackground
        )

        // Detail card (if structured data available)
        if (topic.detailJson != null) {
            try {
                val json = JSONObject(topic.detailJson)
                val type = json.optString("type")
                Surface(
                    color = MaterialTheme.colorScheme.surface,
                    shape = RoundedCornerShape(12.dp),
                    modifier = Modifier.fillMaxWidth()
                ) {
                    Column(modifier = Modifier.padding(16.dp)) {
                        when (type) {
                            "billing" -> {
                                DetailRow("Amount", "₹${json.optString("amount", "—")}")
                                json.optString("due_date", "").takeIf { it.isNotBlank() }?.let { DetailRow("Due Date", it) }
                                json.optString("card_last4", "").takeIf { it.isNotBlank() }?.let { DetailRow("Card", "ending $it") }
                                DetailRow("Status", json.optString("status", "unknown"))
                            }
                            "delivery" -> {
                                DetailRow("Status", json.optString("status", "unknown").replace("_", " "))
                                json.optString("latest_message", "").takeIf { it.isNotBlank() }?.let { DetailRow("Latest", it) }
                            }
                            "calendar" -> {
                                DetailRow("Event", json.optString("event", "—"))
                            }
                        }
                    }
                }
            } catch (_: Exception) {}
        }

        // LLM detail summary
        if (topic.detailSummary != null) {
            Text(
                text = topic.detailSummary,
                style = MaterialTheme.typography.bodyMedium,
                color = MaterialTheme.colorScheme.onSurface.copy(alpha = 0.8f),
                lineHeight = 22.sp
            )
        }

        // Source notifications
        if (state.notifications.isNotEmpty()) {
            Spacer(modifier = Modifier.height(8.dp))
            Text(
                text = "RECEIVED VIA ${topic.channelCount} CHANNEL${if (topic.channelCount > 1) "S" else ""}",
                fontSize = 11.sp,
                fontWeight = FontWeight.Bold,
                color = MaterialTheme.colorScheme.onSurface.copy(alpha = 0.5f)
            )

            state.notifications.forEach { notif ->
                val timeFormat = SimpleDateFormat("h:mm a", Locale.getDefault())
                Surface(
                    color = MaterialTheme.colorScheme.onSurface.copy(alpha = 0.04f),
                    shape = RoundedCornerShape(8.dp),
                    modifier = Modifier.fillMaxWidth()
                ) {
                    Column(modifier = Modifier.padding(10.dp)) {
                        Row(
                            modifier = Modifier.fillMaxWidth(),
                            horizontalArrangement = Arrangement.SpaceBetween
                        ) {
                            Text(
                                text = notif.appName,
                                fontSize = 12.sp,
                                fontWeight = FontWeight.SemiBold,
                                color = MaterialTheme.colorScheme.onBackground
                            )
                            Text(
                                text = timeFormat.format(Date(notif.postedAt)),
                                fontSize = 11.sp,
                                color = MaterialTheme.colorScheme.onSurface.copy(alpha = 0.4f)
                            )
                        }
                        Text(
                            text = notif.content,
                            fontSize = 11.sp,
                            color = MaterialTheme.colorScheme.onSurface.copy(alpha = 0.6f),
                            maxLines = 2,
                            modifier = Modifier.padding(top = 2.dp)
                        )
                    }
                }
            }
        }

        // Action button
        if (topic.actionLabel != null) {
            Spacer(modifier = Modifier.height(8.dp))
            Button(
                onClick = { /* TODO: open actionPackage */ },
                modifier = Modifier.fillMaxWidth(),
                shape = RoundedCornerShape(8.dp)
            ) {
                Text(topic.actionLabel)
            }
        }
    }
}

@Composable
private fun DetailRow(label: String, value: String) {
    Row(
        modifier = Modifier
            .fillMaxWidth()
            .padding(vertical = 4.dp),
        horizontalArrangement = Arrangement.SpaceBetween
    ) {
        Text(text = label, fontSize = 12.sp, color = MaterialTheme.colorScheme.onSurface.copy(alpha = 0.5f))
        Text(text = value, fontSize = 13.sp, fontWeight = FontWeight.SemiBold, color = MaterialTheme.colorScheme.onBackground)
    }
}

package com.focal.ui.navigation

import androidx.compose.foundation.layout.padding
import androidx.compose.material3.Icon
import androidx.compose.material3.MaterialTheme
import androidx.compose.material3.NavigationBar
import androidx.compose.material3.NavigationBarItem

import androidx.compose.material3.Text
import androidx.compose.runtime.Composable
import androidx.compose.runtime.getValue
import androidx.compose.ui.Modifier
import androidx.navigation.NavDestination.Companion.hierarchy
import androidx.navigation.NavGraph.Companion.findStartDestination
import androidx.navigation.NavType
import androidx.navigation.compose.NavHost
import androidx.navigation.compose.composable
import androidx.navigation.compose.currentBackStackEntryAsState
import androidx.navigation.compose.rememberNavController
import androidx.navigation.navArgument
import com.focal.ui.all.AllNotificationsScreen
import com.focal.ui.digest.DigestScreen
import com.focal.ui.digest.TopicDetailScreen

val bottomNavItems = listOf(Screen.Digest, Screen.All, Screen.Settings)

@Composable
fun FocalNavigation() {
    val navController = rememberNavController()
    val navBackStackEntry by navController.currentBackStackEntryAsState()
    val currentDestination = navBackStackEntry?.destination

    Scaffold(
        bottomBar = {
            NavigationBar(
                containerColor = MaterialTheme.colorScheme.surface
            ) {
                bottomNavItems.forEach { screen ->
                    NavigationBarItem(
                        icon = { Icon(screen.icon, contentDescription = screen.title) },
                        label = { Text(screen.title) },
                        selected = currentDestination?.hierarchy?.any { it.route == screen.route } == true,
                        onClick = {
                            navController.navigate(screen.route) {
                                popUpTo(navController.graph.findStartDestination().id) {
                                    saveState = true
                                }
                                launchSingleTop = true
                                restoreState = true
                            }
                        }
                    )
                }
            }
        }
    ) { innerPadding ->
        NavHost(
            navController = navController,
            startDestination = Screen.Digest.route,
            modifier = Modifier.padding(innerPadding)
        ) {
            composable(Screen.Digest.route) {
                DigestScreen(
                    onTopicClick = { topicId ->
                        navController.navigate(Screen.TopicDetail.createRoute(topicId))
                    }
                )
            }
            composable(Screen.All.route) {
                AllNotificationsScreen()
            }
            composable(Screen.Settings.route) {
                Text("Settings — coming soon")
            }
            composable(
                route = Screen.TopicDetail.route,
                arguments = listOf(navArgument("topicId") { type = NavType.StringType })
            ) {
                TopicDetailScreen(
                    onBack = { navController.popBackStack() }
                )
            }
        }
    }
}
