cmake_minimum_required(VERSION 3.22)
project(focal_intelligence)

include(FetchContent)

FetchContent_Declare(
    sentencepiece
    GIT_REPOSITORY https://github.com/google/sentencepiece.git
    GIT_TAG        v0.2.0
)
set(SPM_ENABLE_SHARED   OFF CACHE BOOL "" FORCE)
set(SPM_ENABLE_TCMALLOC OFF CACHE BOOL "" FORCE)
FetchContent_MakeAvailable(sentencepiece)

# SentencePiece's CLI executables (spm_encode, etc.) use Android's liblog via
# their bundled protobuf-lite but don't link -llog themselves. They don't end up
# in the APK (executables are stripped), but they must link successfully.
# sentencepiece_trainer.cc and builder.cc (both in sentencepiece_train-static) emit
# -Winconsistent-missing-override and -Wunused-const-variable under clang/NDK.
# These are upstream issues in SentencePiece v0.2.0; suppress per-target rather
# than globally so our own code still gets full warnings.
if(TARGET sentencepiece_train-static)
    target_compile_options(sentencepiece_train-static PRIVATE
        -Wno-inconsistent-missing-override
        -Wno-unused-const-variable
    )
endif()

foreach(spm_tool spm_encode spm_decode spm_normalize spm_train spm_export_vocab)
    if(TARGET ${spm_tool})
        target_link_libraries(${spm_tool} log)
    endif()
endforeach()

# ---------------------------------------------------------------------------
# LiteRT C API headers — fetched from the v2.1.4 release tarball on GitHub.
# CMake caches the download so subsequent builds are instant.
# LITERT_LIB_DIR is passed in from build.gradle.kts (points to the staged .so).
# ---------------------------------------------------------------------------

FetchContent_Declare(
    litert_headers
    URL      https://github.com/google-ai-edge/LiteRT/archive/refs/tags/v2.1.4.tar.gz
    DOWNLOAD_EXTRACT_TIMESTAMP TRUE
    # Only the c/ and build_common/ subdirs are used — nothing is compiled.
)
FetchContent_GetProperties(litert_headers)
if(NOT litert_headers_POPULATED)
    FetchContent_Populate(litert_headers)
endif()

set(LITERT_HEADERS_DIR "${litert_headers_SOURCE_DIR}")

# build_config.h is generated by LiteRT's own CMake — produce a minimal copy
# so we don't have to build the whole LiteRT project from source.
file(WRITE "${CMAKE_BINARY_DIR}/litert_gen/litert/build_common/build_config.h"
    "#pragma once\n"
    "#define LITERT_BUILD_CONFIG_DISABLE_GPU 0\n"
    "#define LITERT_BUILD_CONFIG_DISABLE_NPU 0\n"
)

if(NOT DEFINED LITERT_LIB_DIR)
    message(FATAL_ERROR "LITERT_LIB_DIR must be set (passed from build.gradle.kts)")
endif()

# Declare libLiteRt.so as a prebuilt imported library for link-time resolution.
# At runtime it is loaded from the APK (packaged by litertlm-android AAR).
# The file at IMPORTED_LOCATION is NOT packaged into the APK by AGP.
add_library(LiteRt SHARED IMPORTED)
set_target_properties(LiteRt PROPERTIES
    IMPORTED_LOCATION "${LITERT_LIB_DIR}/arm64-v8a/libLiteRt.so"
)

add_library(focal_intelligence SHARED
    sentencepiece_jni.cpp
    litert_embedder_jni.cpp
)

target_include_directories(focal_intelligence PRIVATE
    ${sentencepiece_SOURCE_DIR}/src
    ${LITERT_HEADERS_DIR}
    "${CMAKE_BINARY_DIR}/litert_gen"
)

target_link_libraries(focal_intelligence
    sentencepiece-static
    LiteRt
    log
)
